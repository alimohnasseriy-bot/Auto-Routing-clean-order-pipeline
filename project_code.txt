import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

from config.settings import SMALL_FILE_THRESHOLD_MB


def get_file_size_mb(file_path):
    size_bytes = Path(file_path).stat().st_size
    return size_bytes / (1024 * 1024)


def choose_engine(file_path):
    size_mb = get_file_size_mb(file_path)

    if size_mb <= SMALL_FILE_THRESHOLD_MB:
        engine = "python_batch"
        reason = f"file size ({size_mb:.2f} MB) <= {SMALL_FILE_THRESHOLD_MB} MB"
    else:
        engine = "pyspark"
        reason = f"file size ({size_mb:.2f} MB) > {SMALL_FILE_THRESHOLD_MB} MB"

    return size_mb, engine, reason


if __name__ == "__main__":
    if len(sys.argv) != 2:
        print("Usage: python src\\file_router.py <csv_file>")
        sys.exit(1)

    file_path = sys.argv[1]

    if not Path(file_path).exists():
        print(f"ERROR: File not found: {file_path}")
        sys.exit(1)

    size_mb, engine, reason = choose_engine(file_path)

    print("=" * 60)
    print("FILE ROUTER")
    print("=" * 60)
    print(f"File   : {file_path}")
    print(f"Size   : {size_mb:.2f} MB")
    print(f"Limit  : {SMALL_FILE_THRESHOLD_MB} MB")
    print(f"Engine : {engine}")
    print(f"Reason : {reason}")
    print("=" * 60)

import csv
import time
from pathlib import Path

from pymongo import MongoClient

MONGODB_URI = "mongodb://localhost:27017"
DATABASE_NAME = "midterm_data_pipeline"
COLLECTION_NAME = "orders_raw"

BATCH_SIZE = 5000


def load_csv_to_mongodb(file_path, batch_size=BATCH_SIZE):
    client = None

    try:
        client = MongoClient(MONGODB_URI)
        db = client[DATABASE_NAME]
        collection = db[COLLECTION_NAME]

        print("=" * 60)
        print("PYTHON BATCH LOADER")
        print("=" * 60)
        print(f"Input file : {file_path}")
        print(f"Batch size : {batch_size}")
        print(f"Database   : {DATABASE_NAME}")
        print(f"Collection : {COLLECTION_NAME}")
        print("=" * 60)

        total_records = 0
        batch_number = 0
        start_total = time.perf_counter()

        with Path(file_path).open(
            "r",
            encoding="utf-8-sig",
            newline=""
        ) as csv_file:

            reader = csv.DictReader(csv_file)
            batch = []

            for row in reader:
                batch.append(dict(row))

                if len(batch) >= batch_size:
                    batch_number += 1

                    start_batch = time.perf_counter()

                    result = collection.insert_many(
                        batch,
                        ordered=False
                    )

                    elapsed = time.perf_counter() - start_batch

                    inserted = len(result.inserted_ids)
                    total_records += inserted

                    rate = inserted / elapsed if elapsed > 0 else 0

                    print(
                        f"Batch {batch_number:04d} | "
                        f"Records: {inserted:5d} | "
                        f"Time: {elapsed:.3f}s | "
                        f"Rate: {rate:,.0f} records/s"
                    )

                    batch = []

            if batch:
                batch_number += 1

                start_batch = time.perf_counter()

                result = collection.insert_many(
                    batch,
                    ordered=False
                )

                elapsed = time.perf_counter() - start_batch

                inserted = len(result.inserted_ids)
                total_records += inserted

                rate = inserted / elapsed if elapsed > 0 else 0

                print(
                    f"Batch {batch_number:04d} | "
                    f"Records: {inserted:5d} | "
                    f"Time: {elapsed:.3f}s | "
                    f"Rate: {rate:,.0f} records/s"
                )

        total_elapsed = time.perf_counter() - start_total

        overall_rate = (
            total_records / total_elapsed
            if total_elapsed > 0
            else 0
        )

        print("=" * 60)
        print("BATCH LOAD COMPLETED")
        print("=" * 60)
        print(f"Total records : {total_records:,}")
        print(f"Total batches : {batch_number:,}")
        print(f"Total time    : {total_elapsed:.3f}s")
        print(f"Overall rate  : {overall_rate:,.0f} records/s")
        print("=" * 60)

    finally:
        if client is not None:
            client.close()
            print("MongoDB connection closed")


if __name__ == "__main__":
    import argparse

    parser = argparse.ArgumentParser(
        description="Load CSV into MongoDB using streaming batches."
    )

    parser.add_argument(
        "--input",
        default="data/order_huge_mixed_quality.csv",
        help="CSV input file"
    )

    parser.add_argument(
        "--batch-size",
        type=int,
        default=BATCH_SIZE,
        help="Number of records per batch"
    )

    args = parser.parse_args()

    load_csv_to_mongodb(
        args.input,
        args.batch_size
    )

from pymongo import MongoClient
from pymongo.errors import ConnectionFailure, OperationFailure


# ============================================================
# MongoDB Configuration
# ============================================================

MONGO_URI = "mongodb://localhost:27017/"
DATABASE_NAME = "midterm_data_pipeline"


# ============================================================
# Connect to MongoDB
# ============================================================

def connect_to_mongodb():
    try:
        client = MongoClient(
            MONGO_URI,
            serverSelectionTimeoutMS=5000
        )

        # Test connection
        client.admin.command("ping")

        print("âœ“ MongoDB connection successful")

        db = client[DATABASE_NAME]

        return client, db

    except ConnectionFailure as error:
        print("âœ— Failed to connect to MongoDB")
        print(f"Error: {error}")
        raise


# ============================================================
# Create Collections
# ============================================================

def create_collections(db):
    required_collections = [
        "orders_raw",
        "orders_validated",
        "orders_quarantine"
    ]

    existing_collections = db.list_collection_names()

    for collection_name in required_collections:

        if collection_name not in existing_collections:
            db.create_collection(collection_name)
            print(f"âœ“ Collection created: {collection_name}")

        else:
            print(f"âœ“ Collection already exists: {collection_name}")


# ============================================================
# Create Indexes
# ============================================================

def create_indexes(db):

    # Unique index for validated orders
    # id_order is the stable business key
    try:
        db.orders_validated.create_index(
            [("id_order", 1)],
            unique=True,
            name="unique_id_order"
        )

        print("âœ“ Unique index created on orders_validated.id_order")

    except OperationFailure as error:
        print("âœ— Failed to create unique index")
        print(f"Error: {error}")
        raise


# ============================================================
# Main Setup Function
# ============================================================

def setup_database():

    print("=" * 60)
    print("MongoDB Database Setup")
    print("=" * 60)

    client = None

    try:
        # Connect
        client, db = connect_to_mongodb()

        print(f"âœ“ Database: {DATABASE_NAME}")

        # Create collections
        create_collections(db)

        # Create indexes
        create_indexes(db)

        print("=" * 60)
        print("âœ“ MongoDB setup completed successfully")
        print("=" * 60)

        print("\nCollections:")
        for collection in db.list_collection_names():
            print(f"  - {collection}")

        print("\nIndexes on orders_validated:")
        for index in db.orders_validated.list_indexes():
            print(f"  - {index['name']}")

    finally:

        if client:
            client.close()
            print("\nâœ“ MongoDB connection closed")


# ============================================================
# Run
# ============================================================

if __name__ == "__main__":
    setup_database()

from pyspark.sql import SparkSession
from pyspark.sql.functions import col, count, when
import time


INPUT_FILE = "data/orders_huge_mixed_quality.csv"
SPARK_MASTER = "spark://192.168.8.107:7077"

SAMPLE_SIZE = 10000


def main():

    spark = (
        SparkSession.builder
        .appName("LargeFileQualityTest")
        .master(SPARK_MASTER)
        .config("spark.executor.cores", "4")
        .config("spark.executor.memory", "4g")
        .config("spark.sql.shuffle.partitions", "16")
        .getOrCreate()
    )

    spark.sparkContext.setLogLevel("WARN")

    print("=" * 70)
    print("PYSPARK QUALITY TEST")
    print("=" * 70)
    print(f"Input       : {INPUT_FILE}")
    print(f"Master      : {SPARK_MASTER}")
    print(f"Sample size : {SAMPLE_SIZE:,}")
    print("=" * 70)

    start = time.perf_counter()

    df = (
        spark.read
        .option("header", True)
        .option("inferSchema", False)
        .option("multiLine", True)
        .csv(INPUT_FILE)
    )

    print("CSV loaded successfully.")

    sample = df.limit(SAMPLE_SIZE)

    sample_count = sample.count()

    # ---------------------------------------------------------
    # Basic quality checks
    # ---------------------------------------------------------

    stats = sample.select(
        count(
            when(
                col("order_id").isNull() |
                (col("order_id") == ""),
                True
            )
        ).alias("missing_order_id"),

        count(
            when(
                col("customer_email").isNull() |
                (col("customer_email") == ""),
                True
            )
        ).alias("missing_email"),

        count(
            when(
                col("items_json").isNull() |
                (col("items_json") == ""),
                True
            )
        ).alias("missing_items_json"),

        count(
            when(
                col("total_amount").isNull() |
                (col("total_amount") == ""),
                True
            )
        ).alias("missing_total_amount")
    ).collect()[0]

    elapsed = time.perf_counter() - start

    print("=" * 70)
    print("PYSPARK QUALITY TEST COMPLETED")
    print("=" * 70)

    print(f"Sample records     : {sample_count:,}")
    print(f"Missing order_id   : {stats['missing_order_id']:,}")
    print(f"Missing email      : {stats['missing_email']:,}")
    print(f"Missing items_json : {stats['missing_items_json']:,}")
    print(f"Missing total      : {stats['missing_total_amount']:,}")

    print(f"Time               : {elapsed:.3f}s")

    print("=" * 70)

    spark.stop()


if __name__ == "__main__":
    main()

===== quality_processor.py =====
import time
from pymongo import MongoClient

from quality_rules import process_record


MONGODB_URI = "mongodb://localhost:27017"
DATABASE_NAME = "midterm_data_pipeline"

RAW_COLLECTION = "orders_raw"
VALIDATED_COLLECTION = "orders_validated"
QUARANTINE_COLLECTION = "orders_quarantine"


def process_all():
    client = None

    try:
        client = MongoClient(MONGODB_URI)
        db = client[DATABASE_NAME]

        raw = db[RAW_COLLECTION]
        validated = db[VALIDATED_COLLECTION]
        quarantine = db[QUARANTINE_COLLECTION]

        print("=" * 70)
        print("QUALITY PROCESSOR - FULL RUN")
        print("=" * 70)
        print("Processing ALL raw records...")
        print("=" * 70)

        valid_count = 0
        corrected_count = 0
        quarantine_count = 0
        missing_id_count = 0
        duplicate_count = 0

        processed_count = 0

        # Track order_ids seen during this run.
        seen_order_ids = set()

        start = time.perf_counter()

        for record in raw.find():

            processed_count += 1

            original_order_id = record.get("order_id")

            # -------------------------------------------------
            # Missing order_id
            # -------------------------------------------------
            if (
                original_order_id is None
                or str(original_order_id).strip() == ""
            ):
                quarantine_record = dict(record)

                quarantine_record.pop("_id", None)

                quarantine_record["classification"] = "QUARANTINED"
                quarantine_record["quarantine_reasons"] = [
                    "missing_order_id"
                ]
                quarantine_record["quarantine_reason"] = [
                    "missing_order_id"
                ]

                # Use a deterministic key for records without order_id.
                quarantine_key = f"missing-order-id-{processed_count}"

                quarantine.update_one(
                    {"quarantine_key": quarantine_key},
                    {
                        "$set": {
                            **quarantine_record,
                            "quarantine_key": quarantine_key
                        }
                    },
                    upsert=True
                )

                missing_id_count += 1
                quarantine_count += 1

                continue

            order_id = str(original_order_id).strip()

            # -------------------------------------------------
            # Duplicate order_id
            # -------------------------------------------------
            if order_id in seen_order_ids:

                quarantine_record = dict(record)

                quarantine_record.pop("_id", None)

                quarantine_record["classification"] = "QUARANTINED"
                quarantine_record["quarantine_reasons"] = [
                    "duplicate_order_id"
                ]
                quarantine_record["quarantine_reason"] = [
                    "duplicate_order_id"
                ]

                # Each duplicate raw record needs its own key.
                quarantine_key = f"duplicate-{processed_count}-{order_id}"

                quarantine.update_one(
                    {"quarantine_key": quarantine_key},
                    {
                        "$set": {
                            **quarantine_record,
                            "quarantine_key": quarantine_key
                        }
                    },
                    upsert=True
                )

                duplicate_count += 1
                quarantine_count += 1

                continue

            seen_order_ids.add(order_id)

            # -------------------------------------------------
            # Quality rules
            # -------------------------------------------------
            processed, classification = process_record(record)

            processed.pop("_id", None)

            processed["id_order"] = order_id

            # -------------------------------------------------
            # Valid / Corrected
            # -------------------------------------------------
            if classification in ("VALID", "CORRECTED"):

                validated.update_one(
                    {"id_order": order_id},
                    {"$set": processed},
                    upsert=True
                )

                if classification == "VALID":
                    valid_count += 1
                else:
                    corrected_count += 1

            # -------------------------------------------------
            # Quarantined
            # -------------------------------------------------
            elif classification == "QUARANTINED":

                reasons = processed.get(
                    "quarantine_reasons",
                    []
                )

                processed["quarantine_reason"] = reasons

                quarantine.update_one(
                    {"order_id": order_id},
                    {"$set": processed},
                    upsert=True
                )

                quarantine_count += 1

        elapsed = time.perf_counter() - start

        print("=" * 70)
        print("QUALITY PROCESSOR COMPLETED")
        print("=" * 70)

        print(f"Processed             : {processed_count:,}")
        print(f"VALID                 : {valid_count:,}")
        print(f"CORRECTED             : {corrected_count:,}")
        print(f"QUARANTINED           : {quarantine_count:,}")
        print(f"Missing order_id      : {missing_id_count:,}")
        print(f"Duplicate order_id    : {duplicate_count:,}")

        print(f"Time                  : {elapsed:.3f}s")

        rate = (
            processed_count / elapsed
            if elapsed > 0
            else 0
        )

        print(f"Processing rate       : {rate:,.0f} records/s")

        print("=" * 70)

        total_classified = (
            valid_count
            + corrected_count
            + quarantine_count
        )

        print(
            f"Classification check : "
            f"{total_classified:,} / {processed_count:,}"
        )

        if total_classified == processed_count:
            print("Coverage check       : PASS")
        else:
            print("Coverage check       : FAIL")

        print("=" * 70)

    finally:
        if client is not None:
            client.close()
            print("MongoDB connection closed")


if __name__ == "__main__":
    process_all()
